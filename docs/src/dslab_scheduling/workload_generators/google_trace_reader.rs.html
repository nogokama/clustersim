30impl GoogleClusterHostsReader {
31    pub fn read_cluster(&self) -> Vec<HostConfig> {
32        let file = File::open(&self.path).unwrap();
33
34        // Create a CSV reader
35        let mut rdr = ReaderBuilder::new().has_headers(true).from_reader(file);
36
37        let mut records = Vec::new();
38
39        let mut bad_machines = FxHashSet::default();
40        for line in rdr.deserialize() {
41            let record: MachineEvent = line.unwrap();
42
43            if record.time.is_none() {
44                continue;
45            }
46
47            if record.r#type.is_none() && record.machine_id.is_some() {
48                bad_machines.insert(record.machine_id.unwrap());
49            }
50
51            if record.r#type.unwrap() == machine_event::EventType::Unknown {
52                if let Some(machine_id) = record.machine_id {
53                    bad_machines.insert(machine_id);
54                }
55            }
56
57            records.push(record);
58
59            // let record: MachineEvent = line.unwrap();
60            // println!("{:?}", record);
61
62            // records.push(record);
63        }
64
65        records.sort_by(|a, b| a.time.unwrap().cmp(&b.time.unwrap()));
66        let machines = records
67            .iter()
68            .filter(|e| {
69                !bad_machines.contains(&e.machine_id.unwrap())
70                    && e.r#type.unwrap() == machine_event::EventType::Add
71                    && e.cpus.is_some()
72                    && e.memory.is_some()
73            })
74            .map(|e| {
75                HostConfig::from_cpus_memory(
76                    e.machine_id.unwrap() as u64,
77                    (e.cpus.unwrap() * self.resource_multiplier) as u32,
78                    (e.memory.unwrap() * self.resource_multiplier) as u64,
79                )
80            })
81            .collect::<Vec<_>>();
82
83        machines
84    }
85}

140impl GoogleTraceWorkloadGenerator {
141    fn parse_workload(&self) -> Vec<ExecutionDefinition> {
142        let mut reader = ReaderBuilder::new()
143            .has_headers(true)
144            .from_reader(BufReader::new(
145                File::open(&self.instances_path).unwrap_or_else(|_| {
146                    panic!(
147                        "can't find file with instance events: {}",
148                        self.instances_path
149                    )
150                }),
151            ));
152
153        let mut submit_time = FxHashMap::default();
154        let mut schedule_time = FxHashMap::default();
155        let mut finished_time = FxHashMap::default();
156        let mut skip_ids = FxHashSet::default();
157
158        let mut cpus_mapping = FxHashMap::default();
159        let mut memory_mapping = FxHashMap::default();
160
161        let mut events = Vec::new();
162        for line in reader.deserialize() {
163            let record: InstanceEvent = line.unwrap();
164            if record.time.is_some() {
165                events.push(record);
166            }
167        }
168
169        events.sort_by(|a, b| a.time.unwrap().cmp(&b.time.unwrap()));
170
171        for record in events.iter() {
172            if let Some(t) = record.r#type {
173                if record.collection_id.is_none() || record.instance_index.is_none() {
174                    continue;
175                }
176
177                let id = (
178                    record.collection_id.unwrap(),
179                    record.instance_index.unwrap(),
180                );
181                match t {
182                    EventType::Enable => {
183                        if let Some(cpus) = record.cpus {
184                            if let Some(memory) = record.memory {
185                                if let Some(time) = record.time {
186                                    submit_time.entry(id).or_insert_with(|| {
187                                        cpus_mapping.insert(id, cpus);
188                                        memory_mapping.insert(id, memory);
189                                        time
190                                    });
191                                }
192                            }
193                        }
194                    }
195                    EventType::Schedule => {
196                        schedule_time
197                            .entry(id)
198                            .or_insert_with(|| record.time.unwrap());
199                    }
200                    EventType::Finish => {
201                        finished_time.insert(id, record.time.unwrap());
202                    }
203                    EventType::Evict
204                    | EventType::Fail
205                    | EventType::Kill
206                    | EventType::Lost
207                    | EventType::UpdatePending
208                    | EventType::UpdateRunning
209                    | EventType::Queue => {
210                        skip_ids.insert(id);
211                    }
212                    _ => {}
213                }
214            }
215        }
216
217        // println!("cnt: {}", cnt);
218        // println!("submit_time: {}", submit_time.len());
219        // println!("sched time: {}", schedule_time.len());
220        // println!("finished_time: {}", finished_time.len());
221
222        let mut tasks = Vec::new();
223        for (id, t) in finished_time.iter() {
224            if skip_ids.contains(id) {
225                continue;
226            }
227
228            if let Some(s_time) = schedule_time.get(id) {
229                let sched_time = *s_time as f64 / self.time_scale;
230                let finished_time = *t as f64 / self.time_scale;
231                if finished_time < sched_time {
232                    continue;
233                }
234                let cpus = (cpus_mapping.get(id).unwrap() * self.resources_multiplier) as u32;
235                if cpus == 0 {
236                    continue;
237                }
238                let memory = (memory_mapping.get(id).unwrap() * self.resources_multiplier) as u64;
239                let flops = (finished_time - sched_time) * cpus as f64;
240                if flops == 0. {
241                    continue;
242                }
243                tasks.push(ExecutionDefinition {
244                    instance_index: id.1 as u64,
245                    collection_id: id.0 as u64,
246                    cpus,
247                    memory,
248                    submit_time: *submit_time.get(id).unwrap() as f64 / self.time_scale,
249                    flops,
250                    priority: None,
251                });
252            }
253        }
254        // println!("tasks: {}", tasks.len());
255        tasks
256    }

258    pub fn dump_workload_to_native(&self, out_path: &str) {
259        let file = File::create(out_path).unwrap();
260        let writer = BufWriter::new(file);
261
262        let workload = self.parse_workload();
263
264        let converted = workload
265            .iter()
266            .map(|d| NativeExecutionDefinition {
267                id: None,
268                name: None,
269                submit_time: d.submit_time,
270                resources: ResourceRequirements {
271                    nodes_count: 1,
272                    cpu_per_node: d.cpus,
273                    memory_per_node: d.memory,
274                },
275                profile: crate::execution_profiles::builder::ProfileDefinition::Detailed {
276                    r#type: ComputeHomogenous::get_name(),
277                    args: serde_yaml::to_value(ComputeHomogenous {
278                        compute_work: d.flops,
279                    })
280                    .unwrap(),
281                },
282                wall_time_limit: None,
283                priority: d.priority,
284                collection_id: Some(d.collection_id),
285                execution_index: Some(d.instance_index),
286            })
287            .collect::<Vec<_>>();
288
289        serde_yaml::to_writer(writer, &converted).unwrap();
290    }
