19pub struct ClusterHost {

29impl ClusterHost {
30    pub fn new(
31        compute: Rc<RefCell<Compute>>,
32        network: Option<Rc<RefCell<Network>>>,
33        disk: Option<Rc<RefCell<Disk>>>,
34        shared_info_storage: Rc<RefCell<SharedInfoStorage>>,
35        monitoring: Rc<RefCell<Monitoring>>,
36        group_prefix: Option<String>,
37        ctx: SimulationContext,
38    ) -> ClusterHost {
39        ClusterHost {

41            network,
42            disk,
43            shared_info_storage,
44            monitoring,

46            ctx,
47        }

50    pub fn id(&self) -> Id {
51        self.ctx.id()
52    }
53
54    pub async fn sleep(&self, time: f64) {
55        self.ctx.sleep(time).await;
56    }
57
58    pub async fn run_compute(
59        &self,
60        compute_work: f64,
61        compute_allocation_id: u64,
62        cores_dependency: CoresDependency,
63    ) {
64        let req_id = self.compute.borrow_mut().run_on_allocation(
65            compute_work,
66            compute_allocation_id,
67            cores_dependency,
68            self.ctx.id(),
69        );
70
71        log_debug!(
72            self.ctx,
73            "running flops: id={}, flops={}",
74            req_id,
75            compute_work
76        );
77        self.ctx.recv_event_by_key::<CompStarted>(req_id).await;
78
79        self.ctx.recv_event_by_key::<CompFinished>(req_id).await;
80
81        log_debug!(
82            self.ctx,
83            "completed flops: id={}, flops={}",
84            req_id,
85            compute_work
86        );
87    }

89    pub async fn transfer_data_to_process(&self, size: f64, dst_process: ProcessId) {
90        let dst_host = self.shared_info_storage.borrow().get_host_id(dst_process);
91        self.transfer_data(size, self.ctx.id(), dst_host).await;
92    }
93
94    pub async fn transfer_data_from_process(&self, size: f64, src_process: ProcessId) {
95        let src_host = self.shared_info_storage.borrow().get_host_id(src_process);
96        self.transfer_data(size, src_host, self.ctx.id()).await;
97    }

99    pub async fn transfer_data_to_component(&self, size: f64, component_id: Id) {
100        self.transfer_data(size, self.ctx.id(), component_id).await;
101    }
102
103    pub async fn transfer_data_from_component(&self, size: f64, component_id: Id) {
104        self.transfer_data(size, component_id, self.ctx.id()).await;
105    }
106
107    async fn transfer_data(&self, size: f64, src: Id, dst: Id) {
108        let network = self
109            .network
110            .as_ref()
111            .expect("network must be configured to call network operations");
112
113        let req_id = network
114            .borrow_mut()
115            .transfer_data(src, dst, size, self.ctx.id());
116
117        self.ctx
118            .recv_event_by_key::<DataTransferCompleted>(req_id as u64)
119            .await;
120    }
121
122    pub async fn write_data(&self, size: u64) -> Result<(), String> {
123        let req_id = self
124            .disk
125            .as_ref()
126            .expect("disk must be configured to call disk operations")
127            .borrow_mut()
128            .write(size, self.ctx.id());
129
130        select! {
131            _ = self.ctx.recv_event_by_key::<DataWriteCompleted>(req_id).fuse() => {
132                Result::Ok(())
133            }
134            failed = self.ctx.recv_event_by_key::<DataWriteFailed>(req_id).fuse() => {
135                Result::Err(failed.data.error)
136            }
137        }
138    }
139
140    pub async fn read_data(&self, size: u64) -> Result<(), String> {
141        let req_id = self
142            .disk
143            .as_ref()
144            .expect("disk must be configured to call disk operations")
145            .borrow_mut()
146            .read(size, self.ctx.id());
147
148        select! {
149            _ = self.ctx.recv_event_by_key::<DataReadCompleted>(req_id).fuse() => {
150                Result::Ok(())
151            }
152            failed = self.ctx.recv_event_by_key::<DataReadFailed>(req_id).fuse() => {
153                Result::Err(failed.data.error)
154            }
155        }
156    }
157
158    pub fn log_compute_load(&self) {
159        let cpu_used =
160            self.compute.borrow().cores_total() - self.compute.borrow().cores_available();
161        let memory_used =
162            self.compute.borrow().memory_total() - self.compute.borrow().memory_available();
163        self.monitoring.borrow_mut().update_host(
164            self.ctx.time(),
165            self.ctx.name(),
166            self.group_prefix.as_ref(),
167            cpu_used,
168            memory_used,
169            None,
170        );
171    }
172}
