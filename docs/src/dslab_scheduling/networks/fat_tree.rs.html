1use dslab_network::{models::SharedBandwidthNetworkModel, Link, Network};
2use rustc_hash::FxHashMap;
3
4use crate::config::sim_config::NetworkLinkConfig;
5
6pub fn make_fat_tree_topology(
7    network: &mut Network,
8    l2_switch_count: usize,
9    l1_switch_count: usize,
10    switches: FxHashMap<String, usize>, // switch[i] shows the l1 switch number for host i
11    uplink_config: &NetworkLinkConfig,
12    downlink_config: &NetworkLinkConfig,
13    switch_config: &NetworkLinkConfig,
14) {
15    for i in 0..l2_switch_count {
16        let switch_name = format!("l2_switch_{}", i);
17        network.add_node(
18            &switch_name,
19            Box::new(SharedBandwidthNetworkModel::new(
20                switch_config.bandwidth,
21                switch_config.latency,
22            )),
23        );
24    }
25
26    for i in 0..l1_switch_count {
27        let switch_name = format!("l1_switch_{}", i);
28        network.add_node(
29            &switch_name,
30            Box::new(SharedBandwidthNetworkModel::new(
31                switch_config.bandwidth,
32                switch_config.latency,
33            )),
34        );
35
36        for j in 0..l2_switch_count {
37            network.add_link(
38                &switch_name,
39                &format!("l2_switch_{}", j),
40                Link::shared(uplink_config.bandwidth, uplink_config.latency),
41            );
42        }
43    }
44
45    for (host_name, l1_num) in switches.iter() {
46        assert!(
47            *l1_num < l1_switch_count,
48            "Invalid switch number {}",
49            l1_num
50        );
51        let switch_name = format!("l1_switch_{}", l1_num);
52        network.add_link(
53            &switch_name,
54            host_name,
55            Link::shared(downlink_config.bandwidth, downlink_config.latency),
56        );
57    }
58}
